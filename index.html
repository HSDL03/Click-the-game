-- ============================================ -- 1) profiles 表：补齐所有字段 -- ============================================ create table if not exists public.profiles ( id uuid primary key references auth.users(id) on delete cascade, points integer not null default 0, upgrades jsonb not null default '{"autoClick":0}'::jsonb, created_at timestamptz not null default now(), updated_at timestamptz not null default now() ); -- 补字段（无论表是否存在都安全） alter table public.profiles add column if not exists total_clicks bigint not null default 0; alter table public.profiles add column if not exists points integer not null default 0; alter table public.profiles add column if not exists upgrades jsonb not null default '{"autoClick":0}'::jsonb; -- ============================================ -- 2) RLS 策略 -- ============================================ alter table public.profiles enable row level security; drop policy if exists "profiles_select_own" on public.profiles; drop policy if exists "profiles_insert_own" on public.profiles; drop policy if exists "profiles_update_own" on public.profiles; create policy "profiles_select_own" on public.profiles for select to authenticated using (auth.uid() = id); create policy "profiles_insert_own" on public.profiles for insert to authenticated with check (auth.uid() = id); create policy "profiles_update_own" on public.profiles for update to authenticated using (auth.uid() = id) with check (auth.uid() = id); -- ============================================ -- 3) 自动为 new user 创建 profile -- ============================================ create or replace function public.handle_new_user() returns trigger language plpgsql security definer set search_path = public as $$ begin insert into public.profiles (id, points, total_clicks, upgrades) values (new.id, 0, 0, '{"autoClick":0}'::jsonb) on conflict (id) do nothing; return new; end; $$; drop trigger if exists on_auth_user_created on auth.users; create trigger on_auth_user_created after insert on auth.users for each row execute function public.handle_new_user(); -- 给已有用户补 profile insert into public.profiles (id, points, total_clicks, upgrades) select id, 0, 0, '{"autoClick":0}'::jsonb from auth.users on conflict (id) do nothing; -- ============================================ -- 4) sync_progress：累计点击 + 点数 -- ============================================ create or replace function public.sync_progress(p_clicks integer) returns jsonb language plpgsql security definer set search_path = public as $$ declare v_old_total bigint; v_new_total bigint; v_old_points int; v_new_points int; v_gained int; begin if auth.uid() is null then return jsonb_build_object('ok', false, 'error', 'not_authenticated'); end if; if p_clicks < 0 or p_clicks > 1000000 then return jsonb_build_object('ok', false, 'error', 'invalid_value'); end if; insert into public.profiles (id, points, total_clicks, upgrades) values (auth.uid(), 0, 0, '{"autoClick":0}'::jsonb) on conflict (id) do nothing; select total_clicks, points into v_old_total, v_old_points from public.profiles where id = auth.uid() for update; v_old_total := coalesce(v_old_total, 0); v_old_points := coalesce(v_old_points, 0); v_new_total := v_old_total + p_clicks; v_gained := ((v_new_total / 10) - (v_old_total / 10))::int; v_new_points := v_old_points + v_gained; update public.profiles set total_clicks = v_new_total, points = v_new_points, updated_at = now() where id = auth.uid(); return jsonb_build_object( 'ok', true, 'total_clicks', v_new_total, 'points', v_new_points, 'gained', v_gained ); end; $$; grant execute on function public.sync_progress(integer) to authenticated; -- ============================================ -- 5) purchase_upgrade：购买升级 -- ============================================ create or replace function public.purchase_upgrade( p_key text, p_cost integer, p_max integer ) returns jsonb language plpgsql security definer set search_path = public as $$ declare v_level integer; v_points integer; v_new_level integer; begin if auth.uid() is null then return jsonb_build_object('ok', false, 'error', 'not_authenticated'); end if; if p_key not in ('autoClick') then return jsonb_build_object('ok', false, 'error', 'invalid_key'); end if; if p_cost < 0 or p_max < 1 or p_max > 20 then return jsonb_build_object('ok', false, 'error', 'invalid_param'); end if; select points, coalesce((upgrades->>p_key)::int, 0) into v_points, v_level from public.profiles where id = auth.uid() for update; if v_points is null then return jsonb_build_object('ok', false, 'error', 'profile_not_found'); end if; if v_level >= p_max then return jsonb_build_object('ok', false, 'error', 'max_level'); end if; if v_points < p_cost then return jsonb_build_object('ok', false, 'error', 'insufficient'); end if; v_new_level := v_level + 1; update public.profiles set points = points - p_cost, upgrades = jsonb_set( coalesce(upgrades, '{}'::jsonb), array[p_key], to_jsonb(v_new_level) ), updated_at = now() where id = auth.uid(); return jsonb_build_object( 'ok', true, 'points', v_points - p_cost, 'level', v_new_level ); end; $$; grant execute on function public.purchase_upgrade(text, integer, integer) to authenticated; -- ============================================ -- 6) get_leaderboard：排行榜 -- ============================================ create or replace function public.get_leaderboard() returns table(user_id uuid, name text, score bigint) language sql security definer set search_path = public as $$ select p.id as user_id, coalesce(u.raw_user_meta_data->>'username', split_part(u.email, '@', 1)) as name, p.total_clicks as score from public.profiles p join auth.users u on u.id = p.id where p.total_clicks > 0 order by p.total_clicks desc limit 20; $$; grant execute on function public.get_leaderboard() to authenticated, anon;
